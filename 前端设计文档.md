# 校园植物认养与养护系统 - 前端设计文档

## 1. 项目概述

校园植物认养与养护系统是一个基于Vue 3 + Ant Design Vue的Web应用，旨在为校园师生提供植物认养、养护任务管理、健康监测等功能。系统采用组件化开发思想，支持三级角色权限控制，适配电脑端主流浏览器。

## 2. 技术栈

- **前端框架**: Vue 3.3+
- **构建工具**: Vite 4.x
- **UI组件库**: Ant Design Vue 4.x
- **状态管理**: Pinia 2.x
- **路由管理**: Vue Router 4.x
- **HTTP客户端**: Axios 1.x
- **图表库**: ECharts 5.x
- **权限控制**: sa-token
- **CSS预处理器**: Less

```javascript
// utils/auth.js
export const checkPermission = (permission) => {
  const userRole = store.getters['auth/role']
  const permissions = store.getters['permission/permissions']
  return permissions.includes(permission)
}

// 路由守卫
router.beforeEach((to, from, next) => {
  const requiredPermission = to.meta?.permission
  if (requiredPermission && !checkPermission(requiredPermission)) {

    next('/403')
  } else {
    next()
  }
})
```

### 7.2 API请求封装

```javascript
// utils/request.js
import axios from 'axios'
import { message } from 'ant-design-vue'
import { useAuthStore } from '@/store/modules/auth'

const request = axios.create({
  baseURL: import.meta.env.VITE_API_BASE_URL,
  timeout: 10000
})

// 请求拦截器
request.interceptors.request.use(
  config => {
    const authStore = useAuthStore()
    const token = authStore.token
    if (token) {
      config.headers['Authorization'] = `Bearer ${token}`
    }
    return config
  },
  error => {
    return Promise.reject(error)
  }
)

// 响应拦截器
request.interceptors.response.use(
  response => {
    const { code, data, message: msg } = response.data
    if (code === 200) {
      return data
    } else {
      message.error(msg || '请求失败')
      return Promise.reject(new Error(msg || '请求失败'))
    }
  },
  error => {
    if (error.response?.status === 401) {
      const authStore = useAuthStore()
      authStore.logout()
      message.error('登录已过期，请重新登录')
    } else {
      message.error(error.message || '网络错误')
    }
    return Promise.reject(error)
  }
)

export default request
```

### 7.3 API接口示例

```javascript
// api/plant.js
import request from '@/utils/request'

export const plantApi = {
  // 获取植物列表
  getPlantList(params) {
    return request({
      url: '/plant/list',
      method: 'get',
      params
    })
  },

  // 获取植物详情
  getPlantDetail(id) {
    return request({
      url: `/plant/${id}`,
      method: 'get'
    })
  },

  // 添加植物
  addPlant(data) {
    return request({
      url: '/plant',
      method: 'post',
      data
    })
  },

  // 更新植物信息
  updatePlant(id, data) {
    return request({
      url: `/plant/${id}`,
      method: 'put',
      data
    })
  },

  // 删除植物
  deletePlant(id) {
    return request({
      url: `/plant/${id}`,
      method: 'delete'
    })
  },

  // 上传植物图片
  uploadPlantImage(data) {
    return request({
      url: '/plant/upload/image',
      method: 'post',
      data,
      headers: {
        'Content-Type': 'multipart/form-data'
      }
    })
  }
}
```

```javascript
const lazyLoad = {
  mounted(el, binding) {
    const observer = new IntersectionObserver((entries) => {
      entries.forEach(entry => {
        if (entry.isIntersecting) {
          el.src = binding.value
          observer.unobserve(el)
        }
      })
    })
    observer.observe(el)
  }
}
```
